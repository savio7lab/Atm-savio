#include <stdio.h>

int main()
{
    int pin = 1234, enteredPin;
    int choice, count = 0, i;

    float balance = 5000, amount;

    float transAmount[50], transBalance[50];
    int transType[50];

    printf("===== ATM SYSTEM =====\n");

    printf("Enter PIN: ");
    scanf("%d", &enteredPin);

    if (enteredPin == pin)
    {
        printf("\nLogin Successful!\n");
        printf("Current Balance: Rs. %.2f\n", balance);

        for (choice = 0; choice != 4; )
        {
            printf("\n===== ATM MENU =====\n");
            printf("1. Check Balance\n");
            printf("2. Deposit\n");
            printf("3. Withdraw\n");
            printf("4. Exit\n");

            printf("Enter your choice: ");
            scanf("%d", &choice);

            switch(choice)
            {
                case 1:
                    printf("Current Balance: Rs. %.2f\n",
                           balance);
                    break;

                case 2:
                    printf("Enter deposit amount: ");
                    scanf("%f", &amount);

                    if (amount > 0 && count < 50)
                    {
                        balance = balance + amount;

                        transType[count] = 1;
                        transAmount[count] = amount;
                        transBalance[count] = balance;

                        count++;

                        printf("Deposit Successful!\n");
                        printf("New Balance: Rs. %.2f\n",
                               balance);
                    }
                    else
                    {
                        printf("Invalid amount or log full!\n");
                    }
                    break;

                case 3:
                    printf("Enter withdrawal amount: ");
                    scanf("%f", &amount);

                    if (amount > 0 && amount <= balance
                        && count < 50)
                    {
                        balance = balance - amount;

                        transType[count] = 2;
                        transAmount[count] = amount;
                        transBalance[count] = balance;

                        count++;

                        printf("Withdrawal Successful!\n");
                        printf("New Balance: Rs. %.2f\n",
                               balance);
                    }
                    else
                    {
                        printf("Invalid amount, insufficient "
                               "balance or log full!\n");
                    }
                    break;

                case 4:
                    printf("\nThank you for using ATM!\n");
                    break;

                default:
                    printf("Invalid Choice!\n");
            }
        }

        printf("\n===== TRANSACTION HISTORY =====\n");
        printf("S.No  Type       Amount      Balance\n");

        for (i = 0; i < count; i++)
        {
            if (transType[i] == 1)
                printf("%-5d Deposit    ", i + 1);
            else
                printf("%-5d Withdraw   ", i + 1);

            printf("%.2f      %.2f\n",
                   transAmount[i], transBalance[i]);
        }
    }
    else
    {
        printf("Invalid PIN!\n");
    }

    return 0;
}
